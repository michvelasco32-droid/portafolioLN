import re

import nltk
#descarga de modelo de tokenizacion
#nltk.download('punkt_tab')
#importamos las funciones de la libreria
from nltk.tokenize import word_tokenize, sent_tokenize

#ejemplo de texto a tokenizar
texto=("En la materia de lenguaje natural PLN y visión artificial VA, se desarrolla un portfolio"
       "de practicas cómo material de apoyo para los siguientes semestres de la especialidad. "
       "Inculyendo practicas y glosario de funciones por cada practica.")

#como buena practica antes de tokenizar, jay que limpiar el texto
#convertir en minusculas
texto_limpio=texto.lower()

#quitar saltos de linea y espacios
texto_limpio=texto_limpio.strip()

#elimina caracteres que no sean letras
texto_limpio =re.sub(r'[^a-záéíóúüñ\s]', '', texto_limpio)

#TOKENIZAR
#token por palabra
palabras=word_tokenize(texto_limpio, language='spanish')
print("token en palabras: ",palabras)

#token por oraciones
oraciones=sent_tokenize(texto)
print("Token oraciones: ",oraciones)

print(texto)
print(texto_limpio)

"""EL procesamiento de texto en Procesamiento de Lenguaje Natural (PLN), es una etapa esencial
que transforma datos textuales crudos en un formato estructurado, limpio y adecuado para que los
algoritmos de aprendizaje automático puedan analizarlos y procesarlos de manera eficiente. 
Este proceso ayuda a reducir ruido, estandarizar el texto y conservar el significado esencial
del lenguaje."""

📚 Glosario de Funciones

1. sent_tokenize(texto)

Esta funcion separa el texto en oraciones.

2. word_tokenize(texto_limpio, language='spanish')

Esta funcion separa el texto por palabras dependiendo el idioma.

3. re.sub(r'[^a-záéíóúüñ\s]', '', texto_limpio)

Esta funcion elimina los caracteres especiales.
